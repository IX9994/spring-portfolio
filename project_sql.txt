-- ユーザーID採番用（User.java: @SequenceGenerator(sequenceName = "USER_SEQ", allocationSize = 1)）
CREATE SEQUENCE USER_SEQ
    START WITH 1
    INCREMENT BY 1
    NOCACHE
    NOCYCLE;

-- タスクID採番用（Task.java: @SequenceGenerator(sequenceName = "TASK_SEQ", allocationSize = 1)）
CREATE SEQUENCE TASK_SEQ
    START WITH 1
    INCREMENT BY 1
    NOCACHE
    NOCYCLE;

-- ユーザー情報（User.java の @Table(name = "USERS") と各 @Column に対応）
CREATE TABLE USERS (
    ID       NUMBER(10, 0)   NOT NULL,
    EMAIL    VARCHAR2(100 CHAR) NOT NULL,
    PASSWORD VARCHAR2(100 CHAR) NOT NULL,
    CONSTRAINT PK_USERS PRIMARY KEY (ID),
    CONSTRAINT UQ_USERS_EMAIL UNIQUE (EMAIL)
);

-- タスク情報（Task.java の @Table(name = "TASK") と各 @Column に対応）
CREATE TABLE TASK (
    ID           NUMBER(10, 0)    NOT NULL,
    TITLE        VARCHAR2(100 CHAR) NOT NULL,
    DESCRIPTION  VARCHAR2(500 CHAR),
    COMPLETED    NUMBER(10, 0) DEFAULT 0 NOT NULL,
    CREATED_DATE DATE           NOT NULL,
    DUE_DATE     DATE,
    USER_ID      NUMBER(10, 0)  NOT NULL,
    CONSTRAINT PK_TASK PRIMARY KEY (ID),
    CONSTRAINT CK_TASK_COMPLETED CHECK (COMPLETED IN (0, 1)),
    CONSTRAINT FK_TASK_USER FOREIGN KEY (USER_ID) REFERENCES USERS (ID)
);

-- ユーザーごとのタスク一覧・検索を効率化する補助索引
CREATE INDEX IX_TASK_USER_ID ON TASK (USER_ID);

SELECT TABLE_NAME FROM USER_TABLES WHERE TABLE_NAME IN ('USERS', 'TASK') ORDER BY TABLE_NAME;
SELECT SEQUENCE_NAME FROM USER_SEQUENCES WHERE SEQUENCE_NAME IN ('USER_SEQ', 'TASK_SEQ') ORDER BY SEQUENCE_NAME;
